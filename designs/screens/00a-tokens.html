  --text-lead: clamp(1.0rem, 0.95rem + 0.25vw, 1.125rem);       /* 16px → 18px (iOS Safe) */
  --text-body: clamp(0.9375rem, 0.9rem + 0.15vw, 1.0rem);       /* 15px → 16px */

  /* Swiss Grid Rhythm Increments */
