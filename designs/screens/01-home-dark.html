import { pool, telemetry } from "@core/infra";

export async function queryAggregatedMetrics(
  squadId: string,
  windowMs = 3600_000
): Promise<TelemetryReport> {
  return telemetry.trace("metrics.eval", async () => {
    const { rows } = await pool.query(
      `SELECT p95_latency, rpm, error_rate
       FROM mv_stream_telemetry
       WHERE squad_id = $1 AND timestamp > NOW() - $2::interval`,
      [squadId, `${windowMs} milliseconds`]
    );
    return parseTelemetry(rows);
  });
}
